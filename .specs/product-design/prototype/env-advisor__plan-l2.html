# ① 装引擎
brew install llama.cpp

# ② 下模型（含校验）
huggingface-cli download Qwen/Qwen3.5-2B-GGUF Qwen3.5-2B-Q8_0.gguf \
  --local-dir ~/models     # sha256 校验行随命令给出

# ③ 起服务 + ④ 只读校验
llama-server -m ~/models/Qwen3.5-2B-Q8_0.gguf --port 8080
curl -s http://127.0.0.1:8080/v1/models
