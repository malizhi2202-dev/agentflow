命令清单（复制自跑 · 平台不代执行）

# ① 装引擎
pip install mlx mlx-lm

# ② 下模型（源 + 许可证 + 校验值随行）
huggingface-cli download mlx-community/Qwen3.5-2B-bf16 \
  --local-dir ~/models/qwen35-2b-bf16        # Apache-2.0 · sha256 见页面行内

# ③ 起 OpenAI-compatible 端点
python -m mlx_lm server --model ~/models/qwen35-2b-bf16 --port 8080

# ④ 连通校验（只读）
curl -s http://127.0.0.1:8080/v1/models
